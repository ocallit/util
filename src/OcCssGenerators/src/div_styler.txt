    <script>
        document.addEventListener('DOMContentLoaded', () => {
            const modal = document.getElementById('styleEditorModal');
            const styleButtons = document.querySelectorAll('.style-editor-btn');
            let currentTriggerButton = null;

            // Form Inputs
            const form = modal.querySelector('form');
            const inputs = {
                bgColor: document.getElementById('style-bg-color'),
                textColor: document.getElementById('style-text-color'),
                margin: document.getElementById('style-margin'),
                marginUnit: document.getElementById('style-margin-unit'),
                padding: document.getElementById('style-padding'),
                paddingUnit: document.getElementById('style-padding-unit'),
                borderWidth: document.getElementById('style-border-width'),
                borderWidthUnit: document.getElementById('style-border-width-unit'),
                borderStyle: document.getElementById('style-border-style'),
                borderColor: document.getElementById('style-border-color'),
            };

            // Action Buttons
            const applyBtn = document.getElementById('applyStyleBtn');
            const copyBtn = document.getElementById('copyStyleBtn');
            const cancelBtn = document.getElementById('cancelStyleBtn');

            // --- Event Listeners ---

            // Add click listener to all style buttons to open the modal
            styleButtons.forEach(button => {
                button.addEventListener('click', (e) => {
                    currentTriggerButton = e.currentTarget;
                    const targetId = currentTriggerButton.dataset.styleid;
                    const targetElement = document.getElementById(targetId);
                    if (targetElement) {
                        loadStylesIntoModal(targetElement);
                        modal.showModal();
                    }
                });
            });

            // Handle Apply
            applyBtn.addEventListener('click', (e) => {
                e.preventDefault(); // Prevent form submission
                if (!currentTriggerButton) return;
                
                const styleString = generateStyleString();
                const targetId = currentTriggerButton.dataset.styleid;
                const targetElement = document.getElementById(targetId);
                
                if (targetElement) {
                    targetElement.style.cssText = styleString;
                    currentTriggerButton.dataset.style = styleString;
                }
                modal.close();
            });

            // Handle Copy
            copyBtn.addEventListener('click', () => {
                const styleString = generateStyleString();
                navigator.clipboard.writeText(styleString).then(() => {
                    const originalText = copyBtn.textContent;
                    copyBtn.textContent = 'Copied!';
                    setTimeout(() => {
                        copyBtn.textContent = originalText;
                    }, 1500);
                }).catch(err => {
                    console.error('Failed to copy styles: ', err);
                    alert('Failed to copy styles.');
                });
            });

            // Handle Cancel
            cancelBtn.addEventListener('click', () => modal.close());

            // Close modal if user clicks on the backdrop
            modal.addEventListener('click', (e) => {
                if (e.target === modal) {
                    modal.close();
                }
            });


            // --- Helper Functions ---

            /**
             * Gathers values from modal inputs and creates a CSS style string.
             * @returns {string} The CSS style rule string.
             */
            function generateStyleString() {
                const styles = [];
                styles.push(`background-color: ${inputs.bgColor.value};`);
                styles.push(`color: ${inputs.textColor.value};`);

                if (inputs.margin.value) {
                    styles.push(`margin: ${inputs.margin.value}${inputs.marginUnit.value};`);
                }
                if (inputs.padding.value) {
                    styles.push(`padding: ${inputs.padding.value}${inputs.paddingUnit.value};`);
                }
                if (inputs.borderWidth.value > 0 && inputs.borderStyle.value !== 'none') {
                    styles.push(`border: ${inputs.borderWidth.value}${inputs.borderWidthUnit.value} ${inputs.borderStyle.value} ${inputs.borderColor.value};`);
                } else {
                    styles.push(`border: none;`);
                }
                
                return styles.join(' ');
            }

            /**
             * Reads the current styles of a target element and populates the modal inputs.
             * @param {HTMLElement} element The element whose styles to load.
             */
            function loadStylesIntoModal(element) {
                // Reset form to defaults before loading
                form.reset();

                const computedStyle = getComputedStyle(element);

                // Helper to convert RGB to Hex
                const rgbToHex = (rgb) => {
                    if (!rgb || rgb.indexOf('rgb') === -1) return rgb; // Return if already hex or transparent etc.
                    let sep = rgb.indexOf(",") > -1 ? "," : " ";
                    rgb = rgb.substr(4).split(")")[0].split(sep);
                    let r = (+rgb[0]).toString(16),
                        g = (+rgb[1]).toString(16),
                        b = (+rgb[2]).toString(16);
                    if (r.length == 1) r = "0" + r;
                    if (g.length == 1) g = "0" + g;
                    if (b.length == 1) b = "0" + b;
                    return "#" + r + g + b;
                };

                // Load simple color values
                inputs.bgColor.value = rgbToHex(computedStyle.backgroundColor);
                inputs.textColor.value = rgbToHex(computedStyle.color);

                // Load dimension values (margin, padding, border-width)
                const parseDimension = (value) => {
                    const num = parseFloat(value);
                    const unit = value.replace(num, '').trim() || 'px';
                    return { num, unit };
                };
                
                let dim = parseDimension(computedStyle.marginTop); // assuming uniform margin
                inputs.margin.value = dim.num;
                inputs.marginUnit.value = dim.unit;
                
                dim = parseDimension(computedStyle.paddingTop); // assuming uniform padding
                inputs.padding.value = dim.num;
                inputs.paddingUnit.value = dim.unit;
                
                dim = parseDimension(computedStyle.borderTopWidth); // assuming uniform border
                inputs.borderWidth.value = dim.num;
                inputs.borderWidthUnit.value = dim.unit;
                inputs.borderStyle.value = computedStyle.borderTopStyle;
                inputs.borderColor.value = rgbToHex(computedStyle.borderTopColor);
            }
        });
    </script>
